# Examples

## Create and remove events

Click an empty cell to add a one-hour event, click an event to remove it.

<CreateEventsDemo />

```jsx
import { addHours } from 'date-fns';
import { useState } from 'react';
import Calendar from 'react-lightweight-calendar';

// timeDate is 'yyyy-MM-dd HH:mm:ss' in hour-based views and 'yyyy-MM-dd' in MONTH
const toLocalDate = (timeDate) =>
  new Date(timeDate.length > 10 ? timeDate.replace(' ', 'T') : `${timeDate}T09:00:00`);

export default function EditableCalendar() {
  const [events, setEvents] = useState([]);
  const [currentDate, setCurrentDate] = useState(new Date());

  return (
    <Calendar
      data={events}
      currentView="WEEK_TIME"
      currentDate={currentDate}
      setCurrentDate={setCurrentDate}
      activeTimeDateField="startTime-endTime"
      onCellClick={({ timeDate }) => {
        const start = toLocalDate(timeDate);
        setEvents((prev) => [
          ...prev,
          {
            id: crypto.randomUUID(),
            title: 'New event',
            startTime: start.toISOString(),
            endTime: addHours(start, 1).toISOString(),
          },
        ]);
      }}
      onItemClick={(item) => setEvents((prev) => prev.filter((e) => e.id !== item.id))}
    />
  );
}
```

## Month with drill-down

Start with the month and open a day when its number is clicked.

```jsx
export default function DrillDownCalendar({ events }) {
  const [view, setView] = useState('MONTH');
  const [currentDate, setCurrentDate] = useState(new Date());

  return (
    <>
      <div className="toolbar">
        {['MONTH', 'WEEK_TIME', 'DAY'].map((v) => (
          <button key={v} aria-pressed={view === v} onClick={() => setView(v)}>
            {v}
          </button>
        ))}
      </div>
      <Calendar
        data={events}
        currentView={view}
        currentDate={currentDate}
        setCurrentDate={setCurrentDate}
        activeTimeDateField="startTime-endTime"
        onDayNumberClick={(day) => {
          setCurrentDate(day);
          setView('DAY');
        }}
      />
    </>
  );
}
```

## Load events for the visible range

Fetch only what the current view shows. Work out the range from `currentDate` and the
view, and refetch when either changes.

```jsx
import {
  endOfDay, endOfMonth, endOfWeek,
  startOfDay, startOfMonth, startOfWeek,
} from 'date-fns';
import { useEffect, useState } from 'react';
import Calendar from 'react-lightweight-calendar';

const WEEK_STARTS_ON = 1;

function visibleRange(view, date) {
  const d = new Date(date);
  const week = { weekStartsOn: WEEK_STARTS_ON };
  switch (view) {
    case 'MONTH':
      // the month grid also shows the edges of the previous and next month
      return [startOfWeek(startOfMonth(d), week), endOfWeek(endOfMonth(d), week)];
    case 'WEEK':
    case 'WEEK_TIME':
    case 'WEEK_IN_PLACE':
      return [startOfWeek(d, week), endOfWeek(d, week)];
    default:
      return [startOfDay(d), endOfDay(d)];
  }
}

export default function RemoteCalendar({ view = 'WEEK_TIME' }) {
  const [currentDate, setCurrentDate] = useState(new Date());
  const [events, setEvents] = useState([]);

  useEffect(() => {
    const [from, to] = visibleRange(view, currentDate);
    const controller = new AbortController();
    fetch(`/api/events?from=${from.toISOString()}&to=${to.toISOString()}`, {
      signal: controller.signal,
    })
      .then((res) => res.json())
      .then(setEvents)
      .catch(() => {});
    return () => controller.abort();
  }, [view, currentDate]);

  return (
    <Calendar
      data={events}
      currentView={view}
      currentDate={currentDate}
      setCurrentDate={setCurrentDate}
      activeTimeDateField="startTime-endTime"
      weekStartsOn={WEEK_STARTS_ON}
    />
  );
}
```

## Position by a different field

The same tasks shown by when they were created and last updated, instead of when they
are scheduled.

```jsx
const tasks = [
  {
    id: 'T-1',
    title: 'Write docs',
    createdAt: '2025-06-02T08:30:00Z',
    updatedAt: '2025-06-05T16:00:00Z',
    dueAt: '2025-06-09T12:00:00Z',
  },
];

<Calendar data={tasks} currentView="MONTH" activeTimeDateField="createdAt-updatedAt" />;
<Calendar data={tasks} currentView="MONTH" activeTimeDateField="dueAt" />;
```

## Next.js App Router

Fetch on the server, render the calendar in a Client Component.

```tsx filename="app/schedule/page.tsx"
import Schedule from './Schedule';

export default async function SchedulePage() {
  const events = await getEvents(); // your data source
  return <Schedule events={events} />;
}
```

```tsx filename="app/schedule/Schedule.tsx"
'use client';

import dynamic from 'next/dynamic';
import { useState } from 'react';

// Rendered in the browser only: "today" and time zones are not known on the server.
const Calendar = dynamic(() => import('react-lightweight-calendar'), { ssr: false });

export default function Schedule({ events }: { events: Record<string, any>[] }) {
  const [currentDate, setCurrentDate] = useState<string | Date>(new Date());

  return (
    <Calendar
      data={events}
      currentView="WEEK_TIME"
      currentDate={currentDate}
      setCurrentDate={setCurrentDate}
      activeTimeDateField="startTime-endTime"
    />
  );
}
```

## Every prop

A reference configuration with every prop set.

```jsx
import { de } from 'date-fns/locale';
import { useState } from 'react';
import Calendar from 'react-lightweight-calendar';

export default function FullCalendar({ events }) {
  const [currentDate, setCurrentDate] = useState('2025-06-02');
  const [inactiveCells, setInactiveCells] = useState([]);

  return (
    <Calendar
      data={events}
      activeTimeDateField="startTime-endTime"
      currentView="MONTH"
      currentDate={currentDate}
      setCurrentDate={setCurrentDate}
      weekStartsOn={1}
      locale={de}
      todayLabel="Heute"
      timeDateFormat={{ day: 'EEE', hour: 'HH:mm', monthYear: 'LLLL yyyy' }}
      colorDots={[{ color: '#ef4444', text: 'Deadline', date: '2025-06-12' }]}
      cellDisplayMode={{ MONTH: { state: 'CUSTOM', inactiveCells } }}
      disableHoverEffect={false}
      renderItem={(item, isHovered) => (
        <div className={isHovered ? 'event event--hovered' : 'event'}>{item.title}</div>
      )}
      renderItemText={(item) => <span>{item.title}</span>} // ignored while renderItem is set
      renderHeaderItem={(item, { isFromPrevious, isFromNext }) => (
        <div className="all-day">
          {isFromPrevious && '← '}
          {item.title}
          {isFromNext && ' →'}
        </div>
      )}
      renderHeaderItemText={(item) => <span>{item.title}</span>} // ignored while renderHeaderItem is set
      onItemClick={(item, event) => console.log('item', item)}
      onCellClick={(dateInfo) =>
        setInactiveCells((cells) =>
          cells.includes(dateInfo.timeDate)
            ? cells.filter((c) => c !== dateInfo.timeDate)
            : [...cells, dateInfo.timeDate],
        )
      }
      onCellHeaderClick={(dateInfo) => console.log('cell header', dateInfo)}
      onDayNumberClick={(day) => console.log('day number', day)}
      onDayStringClick={(day) => console.log('weekday', day)}
      onHourClick={(value) => console.log('hour', value)}
      onColorDotClick={(dot) => console.log('dot', dot)}
    />
  );
}
```
